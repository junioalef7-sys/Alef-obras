// Estado global da aplicação const state = { obras: [], transacoes: [], equipe: [] }; // Inicialização da Aplicação document.addEventListener('DOMContentLoaded', () => { lucide.createIcons(); loadData(); renderDashboard(); }); // Manipulação do LocalStorage (Persistência) function saveData() { localStorage.setItem('edifiq_db', JSON.stringify(state)); } function loadData() { const data = localStorage.getItem('edifiq_db'); if (data) { Object.assign(state, JSON.parse(data)); } } // Navegação entre telas function navigate(target) { document.querySelectorAll('.nav-btn').forEach(btn => { btn.classList.remove('bg-slate-800', 'text-indigo-400'); btn.classList.add('text-slate-300'); if (btn.dataset.target === target) { btn.classList.add('bg-slate-800', 'text-indigo-400'); } }); const titles = { dashboard: 'Painel Geral', obras: 'Gestão de Obras', equipe: 'Equipe e Diárias', financeiro: 'Controle Financeiro', checklist: 'Checklist de Serviços' }; document.getElementById('page-title').innerText = titles[target] || 'EDIFIQ'; } // Lógica de Renderização do Dashboard function renderDashboard() { document.getElementById('dash-obras-qtd').innerText = state.obras.length; const receita = state.transacoes .filter(t => t.tipo === 'receita') .reduce((acc, curr) => acc + curr.valor, 0); const despesas = state.transacoes .filter(t => t.tipo === 'despesa') .reduce((acc, curr) => acc + curr.valor, 0); const lucro = receita - despesas; document.getElementById('dash-receita').innerText = formatCurrency(receita); document.getElementById('dash-despesas').innerText = formatCurrency(despesas); document.getElementById('dash-lucro').innerText = formatCurrency(lucro); renderTabelaObras(); } function renderTabelaObras() { const tbody = document.getElementById('tbl-obras-recentes'); if (state.obras.length === 0) { tbody.innerHTML = `Nenhuma obra cadastrada.`; return; } tbody.innerHTML = state.obras.map(obra => ` ${obra.nome} ${obra.cliente} Em andamento `).join(''); } // Cadastro de Novas Obras function handleNovaObra(e) { e.preventDefault(); const nome = document.getElementById('obra-nome').value; const cliente = document.getElementById('obra-cliente').value; const novaObra = { id: Date.now().toString(), nome, cliente, dataCriacao: new Date().toISOString() }; state.obras.push(novaObra); saveData(); renderDashboard(); closeModal('nova-obra'); e.target.reset(); } function deletarObra(id) { state.obras = state.obras.filter(o => o.id !== id); saveData(); renderDashboard(); } // Utilitários de Modal function openModal(id) { document.getElementById(`modal-${id}`).classList.remove('hidden'); } function closeModal(id) { document.getElementById(`modal-${id}`).classList.add('hidden'); } // Formatador Monetário function formatCurrency(val) { return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(val); } // Backup de dados via JSON function exportData() { const blob = new Blob([JSON.stringify(state, null, 2)], { type: 'application/json' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = `backup-edifiq-${new Date().toISOString().slice(0, 10)}.json`; a.click(); }
